export async function onRequestPost(context) {
  try {
    const { request, env } = context;
    const { email, passcode, name } = await request.json();

    // Secure server-side call to Resend API (bypasses browser CORS & protects your key)
    const res = await fetch('https://api.resend.com/emails', {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${env.RESEND_API_KEY}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        from: 'Yumilo Market <onboarding@resend.dev>',
        to: [email],
        subject: 'Your Yumilo Market Verification Code',
        html: `
          <div style="font-family: Arial, sans-serif; padding: 20px; color: #333;">
            <h2 style="color: #064e3b;">Yumilo Market Verification</h2>
            <p>Hello ${name || 'Valued User'},</p>
            <p>Your 6-digit verification code is:</p>
            <h1 style="background: #f0fdf4; color: #064e3b; padding: 10px 20px; display: inline-block; letter-spacing: 5px; border-radius: 8px;">${passcode}</h1>
            <p>Enter this code on the website to complete your registration across Zambia.</p>
            <hr style="border: none; border-top: 1px solid #eee; margin: 20px 0;" />
            <p style="font-size: 11px; color: #666;">Zambia's Secure Universal Marketplace Platform</p>
          </div>
        `
      })
    });

    const data = await res.json();

    if (!res.ok) {
      return new Response(JSON.stringify({ success: false, error: data }), {
        status: 400,
        headers: { 'Content-Type': 'application/json' }
      });
    }

    return new Response(JSON.stringify({ success: true, data }), {
      status: 200,
      headers: { 'Content-Type': 'application/json' }
    });
  } catch (err) {
    return new Response(JSON.stringify({ success: false, error: err.message }), {
      status: 500,
      headers: { 'Content-Type': 'application/json' }
    });
  }
}
